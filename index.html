#[inline(always)]
pub fn solve_kinetic_step(
    pos: &mut Vec2,
    vel: &mut Vec2,
    target: Vec2,
    stiffness: f32
) -> Matrix4 {
    // Non-linear spring damping
    let delta = target - *pos;
    let damping = 2.0 * stiffness.sqrt() * 0.88;
    *vel += (delta * stiffness - *vel * damping) * DT;
    *pos += *vel * DT;
    compose_hardware_transform(*pos)
}
